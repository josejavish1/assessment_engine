Localized strings exist exclusively as externalized data artifacts injected at the final compilation boundary.

Unverified vendor marketing ratios and fabricated patch levels (e.g., synthetic sub-versions) are rejected at the ingestion boundary.
5. **Single Source of Truth (SSOT) Mathematical Lock:** The immutable `ActuarialLedger` object generated by the Open FAIR Monte Carlo kernel is the sole provider of financial and risk metrics across the Master Volume Executive Scorecard and all 10 Domain Annexes.
6. **Deterministic Typographical Pacing:** Document layout is governed by an explicit Knuth-Plass-inspired vertical box-and-glue layout engine, guaranteeing an 80% ($\pm 4\%$) target page occupancy, zero orphan headings, zero widow lines, and native script rendering across Latin, Germanic, and CJK character sets.

```python
from __future__ import annotations
from enum import StrEnum
from pathlib import Path
from typing import Annotated, Dict, FrozenSet
from pydantic import BaseModel, ConfigDict, Field, TypeAdapter

class SupportedLocale(StrEnum):
    EN_US = "en_US"
    ES_ES = "es_ES"
    PT_BR = "pt_BR"
    DE_DE = "de_DE"
    FR_FR = "fr_FR"
    JA_JP = "ja_JP"

class ImmutableModel(BaseModel):
    model_config = ConfigDict(
        frozen=True,
        extra="forbid",
        strict=True,
        validate_default=True,
        populate_by_name=False,
    )

class TypographicRules(ImmutableModel):
    decimal_separator: Annotated[str, Field(min_length=1, max_length=1)]
    thousands_separator: Annotated[str, Field(min_length=0, max_length=2)]
    currency_symbol_position: Annotated[str, Field(pattern="^(PREFIX|SUFFIX)$")]
    allow_hyphenation: bool
    line_break_strictness: Annotated[str, Field(pattern="^(normal|strict|anywhere)$")]
    character_expansion_factor: Annotated[float, Field(ge=0.7, le=1.6)]

class LocaleBundle(ImmutableModel):
    locale_id: SupportedLocale
    iso_language_code: Annotated[str, Field(pattern="^[a-z]{2}-[A-Z]{2}$")]
    typography: TypographicRules
    section_headers: Dict[str, str]
    actuarial_terms: Dict[str, str]
    domain_annex_titles: Dict[str, str]
    statutory_citations: Dict[str, str]

class ConfigurationRegistry:
    """Thread-safe, immutable singleton holder for externalized catalogs."""
    def __init__(self, config_root: Path) -> None:
        self._config_root = config_root.resolve(strict=True)
        self._locales: Dict[SupportedLocale, LocaleBundle] = {}
        self._load_all_locales()

    def _load_all_locales(self) -> None:
        locales_dir = self._config_root / "locales"
        adapter = TypeAdapter(LocaleBundle)
        for locale in SupportedLocale:
            file_path = locales_dir / f"{locale.value}.json"
            if not file_path.is_file():
                raise FileNotFoundError(f"Mandatory locale catalog missing: {file_path}")
            self._locales[locale] = adapter.validate_json(file_path.read_bytes())
        self._verify_key_parity()

    def _verify_key_parity(self) -> None:
        baseline = self._locales[SupportedLocale.EN_US]
        baseline_keys: FrozenSet[str] = frozenset(baseline.section_headers.keys())
        for locale, bundle in self._locales.items():
            current_keys = frozenset(bundle.section_headers.keys())
            if current_keys != baseline_keys:
                diff = baseline_keys.symmetric_difference(current_keys)
                raise ValueError(f"Key parity violation in {locale}: {diff}")

    def get_locale(self, locale: SupportedLocale) -> LocaleBundle:
        return self._locales[locale]
```

## PHASE 2: DYNAMIC MULTI-VERTICAL ONTOLOGY & CONTEXT INGESTION ENGINE

### 2.1 Objective
Enable the engine to ingest, normalize, and reason over arbitrary enterprise topologies—from $50M regional manufacturers running air-gapped Modbus/SCADA networks to $100B+ global financial institutions operating hybrid IBM z16 mainframes and multi-region Kubernetes service meshes—while maintaining strict separation between empirical facts and derived estimates.

### 2.2 The Epistemic Provenance Envelope
To enforce Axiom 4 (Absolute Empirical Orthodoxy), raw primitives (`float`, `int`, `str`) are banned from representing infrastructure telemetry in the Domain Layer. Every ingested metric is encapsulated in a generic `ProvenanceEnvelope[T]`.

```python
from datetime import datetime
from enum import StrEnum
from typing import Generic, Optional, TypeVar
from pydantic import Field
from argon.core.config import ImmutableModel

T = TypeVar("T")

class EpistemicStatus(StrEnum):
    EMPIRICAL_VERIFIED = "EMPIRICAL_VERIFIED"          # Direct machine telemetry / config dump
    CLIENT_ATTESTED = "CLIENT_ATTESTED"                # Signed questionnaire / interview log
    ACTUARIALLY_DERIVED = "ACTUARIALLY_DERIVED"        # Posterior output from Bayesian/FAIR engine
    UNVERIFIED_REJECTED = "UNVERIFIED_REJECTED"        # Quarantine state

class ProvenanceEnvelope(ImmutableModel, Generic[T]):
    value: T
    epistemic_status: EpistemicStatus
    source_artifact_sha256: Annotated[str, Field(pattern="^[a-fA-F0-9]{64}$")]
    ingestion_timestamp_utc: datetime
    collector_id: str

    def assert_empirical(self) -> T:

            raise PermissionError(
                f"Operation requires EMPIRICAL_VERIFIED telemetry, got {self.epistemic_status}"

        return self.value
```

### 2.4 Continuous Scale Normalization Function
To prevent false-positive risk inflation in $50M mid-market firms or risk under-estimation in $100B+ conglomerates, structural complexity and financial exposure are normalized using a continuous logarithmic-power scaling tensor:

$$\Phi_{\text{scale}}(R, E, N_{\text{nodes}}) = \left( \frac{R}{R_{\text{ref}}} \right)^{\alpha_v} \cdot \left( 1 + \beta_v \ln\left(\frac{N_{\text{nodes}}}{E}\right) \right) \cdot \Gamma_{\text{topology}}$$

Where:
* $R$: Verified annual gross revenue (normalized to USD via daily ECB/Fed FX catalog rates).
* $R_{\text{ref}}$: Reference baseline revenue ($1.0 \times 10^8$ USD).
* $\alpha_v$: Vertical-specific scaling exponent loaded from `vertical_ontology.yaml`.
* $N_{\text{nodes}}$: Verified count of active compute/network/OT nodes.
* $E$: Verified full-time employee headcount.
* $\beta_v$: Automation density coefficient per vertical.
* $\Gamma_{\text{topology}}$: Graph spectral radius ($\lambda_{\max}$ of the infrastructure adjacency matrix $A$), quantifying blast-radius coupling across Availability Zones, Mainframe LPARs, or SCADA zones.

## PHASE 3: ACTUARIAL RIGOR & PURE OPEN FAIR SIMULATION COUPLING

### 3.1 Objective
Eliminate qualitative "High/Medium/Low" heatmaps and arbitrary scoring heuristics. Implement a mathematically pure, reproducible Bayesian-updated Open FAIR (Factor Analysis of Information Risk) Monte Carlo simulation engine that serves as the Single Source of Truth (SSOT) across all deliverables.

### 3.2 Hierarchical Bayesian Prior Updating (Dealing with Sparse Telemetry)
When client empirical telemetry is partial, the engine does not invent numbers. It computes a formal posterior distribution using industry-vertical priors (derived from VERIS, Advisen, and empirical insurer loss tables stored in `engine_config/priors/empirical_loss_distributions.json`) updated via client-specific empirical evidence.

1. **Threat Event Frequency ($\text{TEF}$):** Modeled as a Gamma-Poisson conjugate hierarchy.
   $$\lambda_{\text{prior}} \sim \text{Gamma}(\alpha_0, \beta_0)$$
   Given $k$ observed threat events over exposure period $T$ in empirical logs (SIEM/EDR/Firewall telemetry):
   $$\lambda_{\text{posterior}} \mid k, T \sim \text{Gamma}(\alpha_0 + k, \, \beta_0 + T)$$

2. **Vulnerability ($\text{Vuln} = P(\text{TC} > \text{RS})$):** Threat Capability ($\text{TC}$) vs. Resistance Strength ($\text{RS}$).
   Resistance Strength is derived from the verified control efficacy matrix $C$ using a Beta-Binomial conjugate model over $n$ audited technical controls (with $s$ passing configurations and $f = n - s$ failing configurations):
   $$\text{RS}_{\text{posterior}} \sim \text{Beta}(\alpha_{\text{control}} + s, \, \beta_{\text{control}} + f)$$

3. **Loss Event Frequency ($\text{LEF}$):**
   $$\text{LEF} = \text{TEF} \times \text{Vuln}$$

### 3.3 Vectorized Quasi-Monte Carlo Loss Magnitude Simulation
For every identified architectural risk scenario $i \in \{1, \dots, M\}$, Loss Magnitude ($\text{LM}_i$) is decomposed strictly into Primary Loss ($\text{PL}_i$) and Secondary Loss ($\text{SL}_i$):

$$\text{PL}_i = L_{\text{productivity}} + L_{\text{response}} + L_{\text{replacement}}$$
$$\text{SL}_i = \mathbb{I}(\text{SLE}_{\text{sec}}) \cdot \left( L_{\text{regulatory\_fines}} + L_{\text{competitive\_advantage}} + L_{\text{reputation\_credit}} \right)$$

* **Primary Loss Distribution:** Modeled via a 3-parameter Pert-Beta or Shifted Log-Normal distribution parameterized by verified Recovery Time Actuals ($\text{RTA}$), Recovery Point Actuals ($\text{RPA}$), and hourly revenue interruption rates.
* **Secondary Heavy-Tail Loss Distribution:** Modeled via a Generalized Pareto Distribution (GPD) or Log-Normal mixture to accurately capture extreme tail risks (e.g., GDPR/NIS2 statutory turnover penalties, SEC disclosure fallout, class-action litigation).
* **Correlated Risk Aggregation (Gaussian / Student-$t$ Copula):** Infrastructure risks are not independent (e.g., an Active Directory compromise correlates with SAN backup encryption and ERP downtime). The engine applies a Student-$t$ copula $C_{\nu, \Sigma}(u_1, \dots, u_M)$ with $\nu = 4$ degrees of freedom and correlation matrix $\Sigma$ derived from the infrastructure dependency graph.

#### Deterministic Sobol Sequence Simulation Kernel (`src/argon/actuarial/open_fair_kernel.py`)
```python
from __future__ import annotations
import hashlib
import numpy as np
from scipy.stats import beta, gamma, lognorm, t as student_t, norm
from scipy.stats.qmc import Sobol
from argon.core.config import ImmutableModel

class RiskScenarioDistributionParams(ImmutableModel):
    scenario_id: str
    domain_annex_id: str
    tef_gamma_alpha: float
    tef_gamma_beta: float
    vuln_beta_alpha: float
    vuln_beta_beta: float
    pl_lognorm_sigma: float
    pl_lognorm_scale: float
    sl_prob_occurrence: float
    sl_lognorm_sigma: float
    sl_lognorm_scale: float

class ScenarioActuarialResult(ImmutableModel):
    scenario_id: str
    domain_annex_id: str
    mean_lef: float
    annualized_loss_expectancy_ale: float
    single_loss_expectancy_mean: float
    single_loss_expectancy_p90: float
    single_loss_expectancy_max_p99_9: float
    loss_exceedance_curve_quantiles: dict[str, float] # e.g., {"0.10": val, "0.50": val, "0.90": val, "0.99": val}

class ImmutableActuarialLedger(ImmutableModel):
    """The Single Source of Truth (SSOT) for all financial metrics across Master & Annexes."""
    ledger_sha256: str
    simulation_iterations: int
    currency_iso: str
    portfolio_total_ale: float
    portfolio_p90_tail_risk: float
    portfolio_p99_value_at_risk: float
    portfolio_max_sle_p99_9: float
    scenario_results: dict[str, ScenarioActuarialResult]
    annex_aggregated_ale: dict[str, float]

class OpenFairMonteCarloEngine:
    """Deterministic Quasi-Monte Carlo Open FAIR Engine using Sobol low-discrepancy sequences."""

    def __init__(self, iterations_power_of_2: int = 16) -> None:
        if not (14 <= iterations_power_of_2 <= 20):
            raise ValueError("Iterations power of 2 must be between 14 (16,384) and 20 (1,048,576).")
        self._m = iterations_power_of_2
        self._n_sims = 2 ** iterations_power_of_2

    def execute_portfolio_simulation(
        self,
        scenarios: list[RiskScenarioDistributionParams],
        correlation_matrix: np.ndarray,
        currency_iso: str,
        seed_ phrase: str,
    ) -> ImmutableActuarialLedger:
        num_scenarios = len(scenarios)
        if correlation_matrix.shape != (num_scenarios, num_scenarios):
            raise ValueError("Correlation matrix dimensions must match scenario count.")

        # Deterministic seed derived from empirical artifact hashes
        seed_int = int(hashlib.sha256(seed_phrase.encode("utf-8")).hexdigest()[:8], 16)
        rng = np.random.default_rng(seed_int)

        # Generate correlated uniform variates via Student-t Copula (df=4) for tail dependence
        cholesky_l = np.linalg.cholesky(correlation_matrix)
        z_norm = rng.standard_normal((self._n_sims, num_scenarios))
        chi2_samples = rng.chisquare(df=4.0, size=(self._n_sims, 1))
        t_samples = (z_norm @ cholesky_l.T) / np.sqrt(chi2_samples / 4.0)
        u_correlated = student_t.cdf(t_samples, df=4.0)

        # Sobol sequence for independent secondary dimensions
        sobol = Sobol(d=num_scenarios * 4, scramble=True, seed=seed_int)
        u_sobol = sobol.random_base2(m=self._m)

        portfolio_annual_losses = np.zeros(self._n_sims, dtype=np.float64)
        scenario_results: dict[str, ScenarioActuarialResult] = {}
        annex_totals: dict[str, float] = {}

        for idx, sc in enumerate(scenarios):
            # 1. Sample LEF = TEF * Vuln using correlated uniform u_correlated[:, idx]
            tef_samples = gamma.ppf(u_correlated[:, idx], a=sc.tef_gamma_alpha, scale=1.0 / sc.tef_gamma_beta)
            vuln_samples = beta.ppf(u_sobol[:, idx * 4], a=sc.vuln_beta_alpha, b=sc.vuln_beta_beta)
            lef_samples = tef_samples * vuln_samples
            event_counts = rng.poisson(lam=lef_samples)

            # 2. Sample Single Loss Expectancy (SLE = Primary + Secondary)
            pl_samples = lognorm.ppf(
                u_sobol[:, idx * 4 + 1], s=sc.pl_lognorm_sigma, scale=sc.pl_lognorm_scale
            )
            sl_trigger = (u_sobol[:, idx * 4 + 2] < sc.sl_prob_occurrence).astype(np.float64)
            sl_samples = sl_trigger * lognorm.ppf(
                u_sobol[:, idx * 4 + 3], s=sc.sl_lognorm_sigma, scale=sc.sl_lognorm_scale
            )
            sle_samples = pl_samples + sl_samples

            # 3. Compound Annual Loss for Scenario idx
            annual_loss_samples = event_counts * sle_samples
            portfolio_annual_losses += annual_loss_samples

            ale_val = float(np.mean(annual_loss_samples))
            sle_mean = float(np.mean(sle_samples))
            sle_p90 = float(np.percentile(sle_samples, 90.0))
            sle_max = float(np.percentile(sle_samples, 99.9))

            quantiles = {
                f"{q:.2f}": float(np.percentile(annual_loss_samples, q * 100.0))
                for q in (0.10, 0.25, 0.50, 0.75, 0.90, 0.95, 0.99)
            }

            scenario_results[sc.scenario_id] = ScenarioActuarialResult(
                scenario_id=sc.scenario_id,
                domain_annex_id=sc.domain_annex_id,
                mean_lef=float(np.mean(lef_samples)),
                annualized_loss_expectancy_ale=round(ale_val, 2),
                single_loss_expectancy_mean=round(sle_mean, 2),
                single_loss_expectancy_p90=round(sle_p90, 2),
                single_loss_expectancy_max_p99_9=round(sle_max, 2),
                loss_exceedance_curve_quantiles=quantiles,
            )
            annex_totals[sc.domain_annex_id] = round(
                annex_totals.get(sc.domain_annex_id, 0.0) + ale_val, 2
            )

        total_ale = round(float(np.sum(list(annex_totals.values()))), 2)
        p90_tail = round(float(np.percentile(portfolio_annual_losses, 90.0)), 2)
        p99_var = round(float(np.percentile(portfolio_annual_losses, 99.0)), 2)
        max_sle_portfolio = round(
            max(r.single_loss_expectancy_max_p99_9 for r in scenario_results.values()), 2
        )

        ledger_payload = f"{total_ale}:{p90_tail}:{p99_var}:{max_sle_portfolio}:{seed_int}"
        ledger_hash = hashlib.sha256(ledger_payload.encode("utf-8")).hexdigest()

        return ImmutableActuarialLedger(
            ledger_sha256=ledger_hash,
            simulation_iterations=self._n_sims,
            currency_iso=currency_iso,
            portfolio_total_ale=total_ale,
            portfolio_p90_tail_risk=p90_tail,
            portfolio_p99_value_at_risk=p99_var,
            portfolio_max_sle_p99_9=max_sle_portfolio,
            scenario_results=scenario_results,
```

### 4.2 Mathematical Vertical Height Budgeting (Knuth-Plass Vertical Solver)
Standard HTML-to-PDF or Markdown-to-PDF pipelines fail because they lack pre-render spatial awareness. Project ARGON introduces a two-pass **Deterministic Vertical Box-and-Glue Solver** (`src/argon/compilers/layout_solver.py`) targeting ISO A4 ($210\text{mm} \times 297\text{mm} = 841.89\text{pt}$ height) and US Letter ($215.9\text{mm} \times 279.4\text{mm} = 792.0\text{pt}$ height).

#### Spatial Equations
For an ISO A4 page with top margin $M_{\text{top}} = 54\text{pt}$ ($19\text{mm}$), bottom margin $M_{\text{bottom}} = 54\text{pt}$ ($19\text{mm}$), running header $H_{\text{hdr}} = 24\text{pt}$, and running footer $H_{\text{ftr}} = 24\text{pt}$:

$$H_{\text{canvas}} = 841.89 - (54 + 54 + 24 + 24) = 685.89\text{pt}$$

Every document block $b_j$ (Paragraph, KPI Grid, Risk Table, Architecture SVG, Callout Box) is modeled as a tuple:
$$b_j = \left( h_j(\mathcal{L}), \, s_j^{+}, \, s_j^{-}, \, p_{\text{break}}(j) \right)$$
Where:
* $h_j(\mathcal{L})$: Deterministic intrinsic height in points under locale $\mathcal{L}$, accounting for the language's `character_expansion_factor` $\xi_{\mathcal{L}}$:
  $$h_{\text{para}}(\mathcal{L}) = \left\lceil \frac{N_{\text{chars}} \cdot \xi_{\mathcal{L}}}{\text{CPL}_{\text{font}}(\mathcal{L})} \right\rceil \cdot \text{Leading}_{\text{pt}} + \text{Margin}_{\text{bottom}}$$
* $s_j^{+}$: Maximum vertical glue stretchability (whitespace expansion tolerance).
* $s_j^{-}$: Maximum vertical glue shrinkability (padding compression tolerance).
* $p_{\text{break}}(j) \in [-\infty, +\infty]$: Penalty for inserting a page break immediately after block $b_j$. Section headings carry $p_{\text{break}} = +\infty$ (prohibiting orphan headers).

#### Page Occupancy Constraint
For every non-terminal page $P_k$ containing blocks $j \in S_k$, the solver enforces the occupancy invariant $\rho(P_k)$:
$$\rho(P_k) = \frac{\sum_{j \in S_k} h_j(\mathcal{L})}{H_{\text{canvas}}} \in [0.76, \, 0.84]$$

If $\rho(P_k) < 0.76$ (under-full page due to a large upcoming table) or $\rho(P_k) > 0.84$ (over-dense wall of text), the solver executes dynamic layout adjustments prior to PDF emission:
1. **Table Row Partitioning with Repeating Headers:** Splits large tables across page boundaries only if at least 3 data rows remain on both sides of the split.
2. **Callout Box Side-by-Side Grid Reflow:** Converts stacked single-column KPI cards into a 2-column CSS Grid/Flexbox arrangement if vertical conservation is required.
3. **Micro-Typographic Tracking & Leading Adjustment:** Adjusts paragraph line-height within $[1.32, 1.45]$ `em` and inter-block spacing within $[8\text{pt}, 16\text{pt}]$.

### 5.3 Production Tribunal Verification Implementation (`src/argon/tribunal/deliverable_auditor.py`)
```python
from __future__ import annotations
from dataclasses import dataclass
from pathlib import Path
import fitz  # PyMuPDF
from argon.actuarial.open_fair_kernel import ImmutableActuarialLedger
from argon.core.config import LocaleBundle

@dataclass(frozen=True)
class TribunalVerdict:
    passed: bool
    gate_name: str
    violations: tuple[str, ...]

class DeliverableTribunalAuditor:
    """Adversarial post-compilation auditor for the Master Volume and 10 Domain Annexes."""

    def __init__(
        self,
        ledger: ImmutableActuarialLedger,
        locale_bundle: LocaleBundle,
        target_occupancy_min: float = 0.72,
        target_occupancy_max: float = 0.88,
    ) -> None:
        self._ledger = ledger
        self._locale = locale_bundle
        self._occ_min = target_occupancy_min
        self._occ_max = target_occupancy_max

    def audit_pdf_geometry_and_ssot(
        self, master_pdf_path: Path, annex_pdf_paths: dict[str, Path]
    ) -> list[TribunalVerdict]:
        verdicts: list[TribunalVerdict] = []
        verdicts.append(self._verify_geometry_invariants(master_pdf_path, is_master=True))
        for annex_id, path in annex_pdf_paths.items():
            verdicts.append(self._verify_geometry_invariants(path, is_master=False))
        verdicts.append(self._verify_cross_document_ssot(master_pdf_path, annex_pdf_paths))
        verdicts.append(self._verify_linguistic_purity(master_pdf_path))
        return verdicts

    def _verify_geometry_invariants(self, pdf_path: Path, is_master: bool) -> TribunalVerdict:
        violations: list[str] = []
        doc = fitz.open(pdf_path)
        try:
            for page_idx in range(len(doc)):
                page = doc[page_idx]
                page_rect = page.rect
                usable_height = page_rect.height - 156.0  # Subtract top/bottom margins + header/footer
                blocks = page.get_text("blocks")

                if not blocks:
                    violations.append(f"{pdf_path.name} Page {page_idx + 1}: Empty page detected.")
                    continue

                # Check for horizontal or vertical clipping out of page bounds
                for b in blocks:
                    x0, y0, x1, y1, text, _, _ = b
                    if x1 > page_rect.width - 36.0 or y1 > page_rect.height - 40.0:
                        snippet = text.strip()[:40].replace("\n", " ")
                        violations.append(
                            f"{pdf_path.name} Page {page_idx + 1}: Bounding box overflow at ({x1:.1f}, {y1:.1f}) -> '{snippet}'"
                        )

                # Check orphan headers (header block in bottom 12% of page without following body block)
                last_block = max(blocks, key=lambda item: item[3])
                last_y0, last_y1, last_text = last_block[1], last_block[3], last_block[4].strip()
                if last_y0 > (page_rect.height * 0.85) and len(last_text) < 90 and last_text.isupper():
                    violations.append(
                        f"{pdf_path.name} Page {page_idx + 1}: Orphan section header at bottom of page -> '{last_text}'"
                    )

                # Check Page 3 Executive Cockpit lock on Master Volume
                if is_master and page_idx == 2:
                    full_page_text = page.get_text("text")
                    expected_cockpit_header = self._locale.section_headers["EXECUTIVE_SCORECARD_TITLE"]
                    if expected_cockpit_header not in full_page_text:
                        violations.append("Master Page 3 missing locked Executive Scorecard title.")
                if is_master and page_idx == 3:
                    page_4_text = page.get_text("text")
                    expected_cockpit_header = self._locale.section_headers["EXECUTIVE_SCORECARD_TITLE"]
                    if expected_cockpit_header in page_4_text:
                        violations.append("CRITICAL: Executive Cockpit spilled over from Page 3 onto Page 4.")
        finally:
            doc.close()

        return TribunalVerdict(
            passed=(len(violations) == 0),
            gate_name=f"GEOMETRY_INVARIANT::{pdf_path.name}",
            violations=tuple(violations),
        )

    def _verify_cross_document_ssot(
        self, master_pdf_path: Path, annex_pdf_paths: dict[str, Path]
    ) -> TribunalVerdict:
        violations: list[str] = []
        master_doc = fitz.open(master_pdf_path)
        try:
            page_3_text = master_doc[2].get_text("text")
            if self._ledger.ledger_sha256[:12] not in page_3_text:
                violations.append(
                    f"Master Page 3 does not bear cryptographic ledger stamp {self._ledger.ledger_sha256[:12]}."
                )
        finally:
            master_doc.close()

        # Verify mathematical identity: Portfolio Total ALE == Sum of all Annex ALEs
        sum_annex_ale = round(sum(self._ledger.annex_aggregated_ale.values()), 2)
        if sum_annex_ale != self._ledger.portfolio_total_ale:
            violations.append(
                f"SSOT Actuarial Divergence: Portfolio ALE ({self._ledger.portfolio_total_ale}) != "
                f"Sum of Annex ALEs ({sum_annex_ale})."
            )

        for annex_id, annex_path in annex_pdf_paths.items():
            annex_doc = fitz.open(annex_path)
            try:
                annex_text = "".join(page.get_text("text") for page in annex_doc)
                if self._ledger.ledger_sha256[:12] not in annex_text:
                    violations.append(
                        f"Annex {annex_id} ({annex_path.name}) missing ledger stamp {self._ledger.ledger_sha256[:12]}."
                    )
            finally:
                annex_doc.close()

        return TribunalVerdict(
            passed=(len(violations) == 0),
            gate_name="ACTUARIAL_SSOT_RECONCILIATION",
            violations=tuple(violations),
        )

    def _verify_linguistic_purity(self, pdf_path: Path) -> TribunalVerdict:
        violations: list[str] = []
        doc = fitz.open(pdf_path)
        try:
            full_text = "\n".join(page.get_text("text") for page in doc)
            # Verify zero unresolved template tokens
            if "{{" in full_text or "}}" in full_text:
                violations.append(f"{pdf_path.name}: Unresolved template token '{{{{...}}}}' detected in output.")
            if "None" in full_text.split() or "NaN" in full_text.split():
                violations.append(f"{pdf_path.name}: Serialized 'None' or 'NaN' literal leaked into output.")
        finally:
            doc.close()

        return TribunalVerdict(
            passed=(len(violations) == 0),
            gate_name=f"LINGUISTIC_PURITY::{pdf_path.name}",
            violations=tuple(violations),
        )
```
